function runPs(script: string, timeout = 60_000): Promise<string> {
  return new Promise((resolve, reject) => {
    execFile(
      'powershell.exe',
      ['-NoProfile', '-NonInteractive', '-Command', psUtf8(script)],
      { timeout },
      (err, stdout, stderr) => {
        if (err) {
          reject(new Error(stderr || err.message || 'Falha desconhecida'))
          return
        }
        resolve(stdout)
      },
    )
  })
}
